# 障碍足迹点（2026-11-08 用户裁定 Q7/Q28 新增）
#
# **为什么不是 OBB**：改造前 `/agc/obstacles` 传的是"连通域 → 旋转外接矩形"，
# 而地面侧真正需要的是"哪些格被占据了"。用 AABB/OBB 表达会引入两重额外误差：
#   * 旋转矩形对 L 形/多栋/斜置实体必然包大（实测框面积/真实足迹 = 1.65×，
#     最差把 workshop_w + wall_w 包成 14×35.5 m，fill=0.30）；
#   * 框的边界随"连通域恰好切在哪里"跳变，不是障碍本身的属性。
# 直接传足迹点，地面侧按格累积（log-odds + carving）即可，表示误差归零。
#
# **为什么新增消息而不是复用 ObstacleTrackArray**：契约 #messages 规定既有 9 个消息的
# 字段与顺序**不得变更**。把点塞进 `ObstacleTrack.pose`（size 置 0）是对既有契约的滥用，
# 会让下游无法区分"一个点"与"一个退化框"。新增第 10 个消息是"加"不是"改"。
#
# 语义：本拍观测到的**障碍像素对应的世界 3D 点**（已按屋顶带过滤掉侧墙主体）。
# 空数组 = 本拍无新观测（不是"障碍消失"），与 TargetTrackArray 的空数组语义同口径。
#
# ---- 点级元数据（ADR-0041，2026-11-10）----
# 几何层不再是"一批没有来历的点"，而是**每个点都带归属与类别**。
# 为什么用**并行数组**而不是嵌套消息：`points` 与屋顶带掩码 `kept` 的"下标一一对应"
# 纪律已经存在（搬点最容易让掩码与点错位），并行数组沿用同一纪律 —— 不需要搬点、
# 不需要嵌套类型，下游按同一个下标读三个数组即可。
# **约定**：并行数组长度必须等于 `points` 长度；全 0/空串表示"本点无该类信息"
# （空数组与全默认值等价，便于旧发布端与单测构造）。

std_msgs/Header header       # 观测时刻（传感器时间）；frame_id=world
	builtin_interfaces/Time stamp
		int32 sec
		uint32 nanosec
	string frame_id
string obstacle_id           # 所属观测批次的标识（跨帧关联前的临时 id，仅供诊断）
geometry_msgs/Point[] points # world 系足迹点（x/y=平面位置，z=该点高度）
	float64 x
	float64 y
	float64 z
float32 confidence           # 本批次观测的声明置信度（当前恒 1.0，无消费者）
string source                # 来源标识（当前恒 "nadir_segmentation"）

# 每个点所属的**障碍实例 id**（与 /agc/obstacles 的 obstacle_id 同一命名空间，
# 形如 "nadir_7"）。空串 = 该点尚未归属任何实例（跨帧关联失败 / 被面积门限滤掉）。
string[] point_instance_id
# 每个点的**类别**（决定它该不该进几何足迹）：
#   0 = 未分类（旧发布端 / 无法判定）
#   1 = 屋顶（顶面 —— 这就是障碍的足迹，应当进地图）
#   2 = 墙 / 侧墙（投影到地面会外扩 H/(H−h) 倍，**不该**当地面足迹）
#   3 = 边缘（深度/分割边界，落位不可信）
#   4 = 遮挡区（被其他障碍挡住，不能当作"看到是空的"）
uint8[] point_class

# ---- 逐点不确定度与观测视角（ADR-0042，2026-11-11）----
# **为什么需要**：此前几何层只知道"这个点在 (x,y)"，不知道"这个点有多准、从哪个方向看的"。
# 于是一个 14 m 高、画幅最外圈、深度差 3% 的点，与一个正下方、深度差 0.1% 的点，
# 在地图里权重完全相同 —— 都是"一个 hit"。四项误差源（位姿位置 / 位姿偏航 /
# 深度轴向 / 像素落位）各自解析传播，方差相加，得到 2×2 协方差。
#
# **为什么是 3 个 float 而不是 4 个**：2×2 对称矩阵只有 xx / xy / yy 三个独立量。
# **为什么是并行数组而不是 geometry_msgs/PoseWithCovariance**：那是 36 个 double，
# 176k 点 = 50 MB/帧，而实际只有平面 3 个分量非零（正下视不产生 z 方向的地面落位误差）。
#
# **约定**：与 `points` **同下标**、长度相等；空数组 = 本发布端不提供该信息
# （等价于"全 0 协方差" = 无信息，下游**不得**当成"非常准"）。

# 每个点位置的 2×2 协方差上三角（world XY，单位 m²）：
#   xx = Var(x), xy = Cov(x,y), yy = Var(y)
# 下游用最大特征值开方得到"最坏方向 1σ"；本仓 `map_adapter` 用它做半格门控。
float32[] point_cov_xx
float32[] point_cov_xy
float32[] point_cov_yy

# 每个点的**观测视角质量**：射线与水平面法向夹角的余弦（1 = 正下方直视，→0 = 掠射）。
# 掠射时同样的深度误差摊到地面上更长，落位更不可信。0 = 无法判定。
float32[] point_view_cos

# ---- 批次级观测元数据（一次曝光共享，故不逐点存）----
# **为什么不把观测时间做成逐点数组**：一帧正下视是**一次曝光**，176k 个点共享同一时刻；
# 逐点存 176k 份相同时间戳是纯浪费。真正缺失的是"标签帧与深度帧差了多少" ——
# 那是批次级量，决定本帧是否存在系统性整体位移（补偿量见 ADR-0039）。
#
# 标签帧与深度帧的时间差（s）。非 0 表示投影端做过空间补偿。
float32 depth_time_offset_s
# 相机世界位置（观测视角起点）。下游可用它算每点的实际入射几何，
# 也可用于判断"这帧是不是正下视"。
geometry_msgs/Point observer_origin
	float64 x
	float64 y
	float64 z
